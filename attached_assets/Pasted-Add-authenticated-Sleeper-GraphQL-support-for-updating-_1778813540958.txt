Add authenticated Sleeper GraphQL support for updating the live draft queue.

Goal:
In the live draft room, users should be able to add/remove/reorder players in their Sleeper draft queue from inside Dynasty Command.

Discovered Sleeper operation:
POST https://sleeper.com/graphql

operationName: update_draft_queue

Mutation:
mutation update_draft_queue($draft_id: ID!, $player_ids: [String!]!) {
  update_draft_queue(player_ids: $player_ids, draft_id: $draft_id)
}

Captured example:
update_draft_queue(player_ids: ["11435"], draft_id: "1312021297428971520")

Response:
{
  "data": {
    "update_draft_queue": ["11435"]
  }
}

Important behavior:
This mutation appears to replace the full queue with the provided player_ids array, not simply append one player.

So:
- To add a player: send existingQueue + newPlayerId
- To remove a player: send existingQueue without that playerId
- To reorder players: send the reordered full player_ids array

Do NOT send only the new player unless you intend to replace the entire queue.

Important auth:
Sleeper’s browser request sends:
authorization: <JWT>
not:
authorization: Bearer <JWT>

Never log or expose the token.

1. Add mutation file.

Create:
src/lib/sleeper/graphql/mutations/updateDraftQueue.ts

Add:

import { sleeperGraphql } from "../client";

export const UPDATE_DRAFT_QUEUE_MUTATION = `
  mutation update_draft_queue($draft_id: ID!, $player_ids: [String!]!) {
    update_draft_queue(player_ids: $player_ids, draft_id: $draft_id)
  }
`;

export async function updateDraftQueue(
  token: string,
  draftId: string,
  playerIds: string[]
) {
  return sleeperGraphql(token, {
    operationName: "update_draft_queue",
    variables: {
      draft_id: draftId,
      player_ids: playerIds,
    },
    query: UPDATE_DRAFT_QUEUE_MUTATION,
  });
}

3. Add queue helper functions.

Create:
src/lib/sleeper/draftQueueActions.ts

Add helper functions:

import { updateDraftQueue } from "./graphql/mutations/updateDraftQueue";

export async function addPlayerToDraftQueue({
  token,
  draftId,
  currentQueue,
  playerId,
}: {
  token: string;
  draftId: string;
  currentQueue: string[];
  playerId: string;
}) {
  if (currentQueue.includes(playerId)) {
    return currentQueue;
  }

  const nextQueue = [...currentQueue, playerId];

  const data = await updateDraftQueue(token, draftId, nextQueue);
  return data.update_draft_queue;
}

export async function removePlayerFromDraftQueue({
  token,
  draftId,
  currentQueue,
  playerId,
}: {
  token: string;
  draftId: string;
  currentQueue: string[];
  playerId: string;
}) {
  const nextQueue = currentQueue.filter((id) => id !== playerId);

  const data = await updateDraftQueue(token, draftId, nextQueue);
  return data.update_draft_queue;
}

export async function reorderDraftQueue({
  token,
  draftId,
  nextQueue,
}: {
  token: string;
  draftId: string;
  nextQueue: string[];
}) {
  const data = await updateDraftQueue(token, draftId, nextQueue);
  return data.update_draft_queue;
}
